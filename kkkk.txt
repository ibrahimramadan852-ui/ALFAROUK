    <script>
        const DEFAULT_STATE = {
            units: [
                { number: "101", ownerName: "أحمد محمد", tenantName: "أحمد محمد", type: "سكني", monthlyFee: 150, arrears: 0 },
                { number: "102", ownerName: "محمود علي", tenantName: "خالد حسن", type: "سكني", monthlyFee: 150, arrears: 300 }
            ],
            expenses: [
                { id: 1, date: "2026/05/12", title: "صيانة إنارة السلم الرئيسي", category: "صيانة", amount: 450 }
            ],
            maintenance: [
                { id: 1, date: "2026/06/05", title: "غسيل وتطهير خزانات المياه العلوية" }
            ],
            logs: [{ timestamp: new Date().toLocaleString('ar-EG'), action: "تم تشغيل نظام اتحاد الشاغلين وتفعيله كتطبيق مستقل." }],
            initBalances: { cash: 5000, bank: 45000 }
        };

        let state = DEFAULT_STATE;

        function loadState() {
            const saved = localStorage.getItem('al_farouk_diamond_state');
            if (saved) {
                try {
                    state = JSON.parse(saved);
                } catch(e) {
                    state = DEFAULT_STATE;
                }
            } else {
                state = DEFAULT_STATE;
                saveState();
            }
        }

        function saveState() {
            localStorage.setItem('al_farouk_diamond_state', JSON.stringify(state));
        }

        function logAction(actionText) {
            state.logs.push({ timestamp: new Date().toLocaleString('ar-EG'), action: actionText });
            saveState();
            renderLogs();
        }

        function login() {
            document.getElementById('loginScreen').classList.add('hidden');
            document.getElementById('app').classList.remove('hidden');
            loadState();
            updateUI();
        }

        function logout() {
            document.getElementById('loginScreen').classList.remove('hidden');
            document.getElementById('app').classList.add('hidden');
        }

        function showTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));
            document.querySelectorAll('.tab-btn').forEach(el => el.classList.remove('bg-blue-600', 'text-white'));
            document.getElementById(tabId + 'Tab').classList.add('active');
            event.currentTarget.classList.add('bg-blue-600', 'text-white');
        }

        function updateUI() {
            renderUnitsTable();
            renderExpensesTable();
            renderMaintenanceTable();
            renderLogs();
            calculateFinancials();
        }

        function renderUnitsTable(filterText = '') {
            const tbody = document.getElementById('unitsTableBody');
            tbody.innerHTML = '';
            state.units.filter(u => u.number.includes(filterText) || u.ownerName.includes(filterText)).forEach(u => {
                tbody.innerHTML += `
                    <tr class="border-b hover:bg-gray-50">
                        <td class="p-2 font-bold">${u.number}</td>
                        <td class="p-2">${u.ownerName}</td>
                        <td class="p-2">${u.tenantName}</td>
                        <td class="p-2">${u.type}</td>
                        <td class="p-2 text-blue-700">${u.monthlyFee} ج.م</td>
                        <td class="p-2 ${u.arrears > 0 ? 'text-red-600 font-bold' : 'text-green-600'}">${u.arrears} ج.م</td>
                        <td class="p-2"><button onclick="deleteUnit('${u.number}')" class="text-red-500">🗑️</button></td>
                    </tr>`;
            });
        }

        function searchUnits() {
            renderUnitsTable(document.getElementById('unitSearch').value);
        }

        function addOrUpdateUnit() {
            const num = document.getElementById('unitNum').value;
            const owner = document.getElementById('unitOwner').value;
            if(!num || !owner) return alert('الرجاء إدخال البيانات الأساسية للشقة.');
            
            const existingIdx = state.units.findIndex(u => u.number === num);
            const unitData = {
                number: num, ownerName: owner, tenantName: document.getElementById('unitTenant').value || owner,
                type: document.getElementById('unitType').value,
                monthlyFee: parseFloat(document.getElementById('unitFee').value) || 0,
                arrears: parseFloat(document.getElementById('unitArrears').value) || 0
            };

            if(existingIdx > -1) {
                state.units[existingIdx] = unitData;
                logAction(`تم تحديث بيانات الشقة رقم ${num}`);
            } else {
                state.units.push(unitData);
                logAction(`تم إضافة وحدة جديدة برقم ${num}`);
            }
            saveState(); updateUI();
            ['unitNum', 'unitOwner', 'unitTenant', 'unitFee', 'unitArrears'].forEach(id => document.getElementById(id).value = '');
        }

        function deleteUnit(num) {
            if(confirm('هل تود حذف هذه الوحدة؟')) {
                state.units = state.units.filter(u => u.number !== num);
                logAction(`تم حذف الوحدة رقم ${num}`);
                saveState(); updateUI();
            }
        }

        function renderExpensesTable() {
            document.getElementById('expensesTableBody').innerHTML = state.expenses.slice().reverse().map(e => `
                <tr class="border-b">
                    <td class="p-2 text-gray-400">${e.date}</td>
                    <td class="p-2 font-medium">${e.title}</td>
                    <td class="p-2"><span class="px-2 py-0.5 bg-gray-100 rounded text-xs">${e.category}</span></td>
                    <td class="p-2 text-red-600 font-bold">${e.amount} ج.م</td>
                    <td class="p-2"><button onclick="deleteExpense(${e.id})" class="text-red-500">🗑️</button></td>
                </tr>`).join('');
        }

        function addExpense() {
            const title = document.getElementById('expTitle').value;
            const amount = parseFloat(document.getElementById('expAmount').value) || 0;
            if(!title || amount <= 0) return alert('برجاء مراجعة بيانات المصروف.');

            state.expenses.push({ id: Date.now(), date: new Date().toLocaleDateString('ar-EG'), title, category: document.getElementById('expCategory').value, amount });
            logAction(`تم تسجيل مصروف جديد: ${title} بقيمة ${amount} ج.م`);
            saveState(); updateUI();
            document.getElementById('expTitle').value = ''; document.getElementById('expAmount').value = '';
        }

        function deleteExpense(id) {
            state.expenses = state.expenses.filter(e => e.id !== id);
            logAction('تم حذف بند مصروفات من المنظومة.');
            saveState(); updateUI();
        }

        function renderMaintenanceTable() {
            document.getElementById('maintenanceTableBody').innerHTML = state.maintenance.slice().reverse().map(m => `
                <tr class="border-b">
                    <td class="p-2">${m.date}</td>
                    <td class="p-2 font-medium">${m.title}</td>
                    <td class="p-2"><button onclick="deleteMaint(${m.id})" class="text-red-500">🗑️</button></td>
                </tr>`).join('');
        }

        function saveMaintenance() {
            const title = document.getElementById('maintTitle').value;
            if(!title) return;
            state.maintenance.push({ id: Date.now(), date: new Date().toLocaleDateString('ar-EG'), title });
            logAction(`تم جدولة مهمة صيانة: ${title}`);
            document.getElementById('maintTitle').value = '';
            saveState(); updateUI();
        }

        function deleteMaint(id) {
            state.maintenance = state.maintenance.filter(m => m.id !== id);
            saveState(); updateUI();
        }

        function renderLogs() {
            document.getElementById('logsContainer').innerHTML = state.logs.slice().reverse().map(l => `
                <div class="p-2 bg-gray-50 border-r-2 border-blue-600 rounded text-[11px]">
                    <span class="text-gray-400 font-mono">[${l.timestamp}]</span> ${l.action}
                </div>`).join('');
        }

        function clearLogs() {
            state.logs = [{ timestamp: new Date().toLocaleString('ar-EG'), action: "تم تصفير سجل النشاط." }];
            saveState(); renderLogs();
        }

        function calculateFinancials() {
            let arrears = 0; state.units.forEach(u => arrears += (u.arrears || 0));
            let expenses = 0; state.expenses.forEach(e => expenses += e.amount);

            document.getElementById('initCash').value = state.initBalances.cash;
            document.getElementById('initBank').value = state.initBalances.bank;

            let currentCash = state.initBalances.cash - (expenses * 0.4);
            let currentBank = state.initBalances.bank - (expenses * 0.6);

            document.getElementById('cashBalance').innerText = `${currentCash.toFixed(0)} ج.م`;
            document.getElementById('bankBalance').innerText = `${currentBank.toFixed(0)} ج.م`;
            document.getElementById('totalExpenses').innerText = `${expenses} ج.م`;
            document.getElementById('totalArrears').innerText = `${arrears} ج.م`;

            renderChart(currentCash, currentBank, expenses, arrears);
        }

        function updateInitBalances() {
            state.initBalances.cash = parseFloat(document.getElementById('initCash').value) || 0;
            state.initBalances.bank = parseFloat(document.getElementById('initBank').value) || 0;
            logAction("تم تحديث الأرصدة الحسابية الافتتاحية للمشروع.");
            saveState(); calculateFinancials();
        }

        let myChart = null;
        function renderChart(cash, bank, exp, arr) {
            const ctx = document.getElementById('financialChart');
            if(!ctx) return;
            if(myChart) myChart.destroy();
            myChart = new Chart(ctx, {
                type: 'bar',
                data: {
                    labels: ['خزينة', 'بنك', 'مصروفات', 'متأخرات'],
                    datasets: [{
                        data: [cash, bank, exp, arr],
                        backgroundColor: ['#3b82f6', '#6366f1', '#ef4444', '#f97316'],
                        borderRadius: 6
                    }]
                },
                options: { responsive: true, plugins: { legend: { display: false } } }
            });
        }

        function exportToExcel() {
            let csv = "\ufeffرقم الشقة,المالك,الساكن,النوع,الاشتراك,المتأخرات\n";
            state.units.forEach(u => { csv += `${u.number},${u.ownerName},${u.tenantName},${u.type},${u.monthlyFee},${u.arrears}\n`; });
            const link = document.createElement("a");
            link.href = URL.createObjectURL(new Blob([csv], { type: 'text/csv;charset=utf-8;' }));
            link.download = "تقرير_اتحاد_الشاغلين.csv"; link.click();
        }

        function exportSystemData() {
            const dataStr = btoa(unescape(encodeURIComponent(JSON.stringify(state))));
            const box = document.getElementById('syncBox'); box.classList.remove('hidden');
            const txt = document.getElementById('syncTextArea'); txt.value = dataStr; txt.select();
            
            const btn = document.getElementById('syncConfirmBtn');
            btn.className = "mt-2 bg-blue-600 text-white px-4 py-2 rounded-lg text-xs font-bold w-full";
            btn.innerText = "نسخ كود المزامنة التلقائي";
            btn.onclick = function() {
                txt.select(); document.execCommand('copy');
                alert('تم نسخ كود التصدير! أرسله لأي جهاز وقم بعمل استيراد هناك.');
            };
        }

        function importSystemData() {
            const box = document.getElementById('syncBox'); box.classList.remove('hidden');
            const txt = document.getElementById('syncTextArea'); txt.value = '';
            
            const btn = document.getElementById('syncConfirmBtn');
            btn.className = "mt-2 bg-green-600 text-white px-4 py-2 rounded-lg text-xs font-bold w-full";
            btn.innerText = "تأكيد دمج واستيراد البيانات للبرنامج";
            btn.onclick = function() {
                try {
                    const decoded = JSON.parse(decodeURIComponent(escape(atob(txt.value.trim()))));
                    if(decoded.units) {
                        state = decoded; saveState(); updateUI();
                        alert('تمت مزامنة البيانات وتحديث التطبيق بنجاح!');
                        box.classList.add('hidden');
                    }
                } catch(e) { alert('الكود غير صحيح أو تالف.'); }
            };
        }
